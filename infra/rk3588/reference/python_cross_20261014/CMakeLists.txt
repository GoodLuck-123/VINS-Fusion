cmake_minimum_required(VERSION 3.16)
project(arm_python_smoke LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development NumPy)
if(NOT Python3_EXECUTABLE STREQUAL "/usr/bin/python3")
  message(FATAL_ERROR "Expected host system Python")
endif()
if(NOT Python3_INCLUDE_DIRS MATCHES "$ENV{VINS_ARM_SYSROOT}" OR NOT Python3_LIBRARIES MATCHES "$ENV{VINS_ARM_SYSROOT}")
  message(FATAL_ERROR "Expected target Python development files")
endif()
Python3_add_library(arm_python_smoke MODULE smoke.cpp)
target_link_libraries(arm_python_smoke PRIVATE Python3::NumPy)
message(STATUS "HOST_INTERPRETER=${Python3_EXECUTABLE}; TARGET_LIBRARY=${Python3_LIBRARIES}")
