cmake_minimum_required(VERSION 3.16)
project(rk_shared_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV 4.2 REQUIRED COMPONENTS core)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(shared_probe main.cpp "${repo}/vins_estimator/src/shared_frames.cpp")
target_compile_features(shared_probe PRIVATE cxx_std_17)
# This probe uses assertions, including syscall checks, in Release too.
target_compile_options(shared_probe PRIVATE -UNDEBUG)
target_include_directories(shared_probe PRIVATE "${repo}/vins_estimator/include" ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(shared_probe cv_bridge)
target_link_libraries(shared_probe ${OpenCV_LIBS} OpenSSL::Crypto Threads::Threads)
