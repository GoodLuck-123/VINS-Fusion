cmake_minimum_required(VERSION 3.16)
project(vins_native_probes)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV 4.2.0 EXACT REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(native_image_probe native_image_probe.cpp)
add_executable(native_shared_probe native_shared_probe.cpp ../vins_estimator/src/shared_frames.cpp)
foreach(target native_image_probe native_shared_probe)
  target_include_directories(${target} PRIVATE ../vins_estimator/include ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(${target} cv_bridge sensor_msgs)
  target_compile_options(${target} PRIVATE -O1 -g -UNDEBUG -fsanitize=address -fno-omit-frame-pointer -fno-pie)
  target_link_options(${target} PRIVATE -fsanitize=address -no-pie)
  target_link_libraries(${target} ${OpenCV_LIBS} OpenSSL::Crypto Threads::Threads)
endforeach()
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(camera_models REQUIRED)
find_package(Ceres 1.14.0 EXACT REQUIRED)
add_executable(native_feature_probe native_feature_probe.cpp)
target_include_directories(native_feature_probe PRIVATE ../vins_estimator/include ../ros2/include "$ENV{CONDA_PREFIX}/include" ${OpenCV_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
ament_target_dependencies(native_feature_probe rclcpp cv_bridge sensor_msgs camera_models)
target_compile_options(native_feature_probe PRIVATE -O1 -g -UNDEBUG -fsanitize=address -fno-omit-frame-pointer -fno-pie)
target_link_options(native_feature_probe PRIVATE -fsanitize=address -no-pie)
target_link_libraries(native_feature_probe ${OpenCV_LIBS} Eigen3::Eigen "${CMAKE_CURRENT_SOURCE_DIR}/../../.infra/ros2_ws/install/lib/libvins_component.so")
